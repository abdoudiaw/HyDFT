cmake_minimum_required(VERSION 3.20)
project(hydft VERSION 0.1.0 LANGUAGES Fortran)

set(CMAKE_Fortran_STANDARD 2018)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
find_package(FFTW REQUIRED)
option(HYDFT_OPENMP "Build with OpenMP" OFF)
if(HYDFT_OPENMP)
  find_package(OpenMP REQUIRED)
endif()

if(CMAKE_Fortran_COMPILER_ID MATCHES "GNU")
  set(CMAKE_Fortran_FLAGS "${CMAKE_Fortran_FLAGS} -ffree-line-length-none -std=f2018 -Wall -Wextra -Wno-unused-dummy-argument -Wno-compare-reals -fimplicit-none")
  set(CMAKE_Fortran_FLAGS_RELEASE "-O3 -march=native")
  set(CMAKE_Fortran_FLAGS_DEBUG "-O0 -g -fcheck=all -fbacktrace -ffpe-trap=invalid,zero,overflow")
elseif(CMAKE_Fortran_COMPILER_ID MATCHES "Intel")
  set(CMAKE_Fortran_FLAGS_RELEASE "-O3 -xHost")
  set(CMAKE_Fortran_FLAGS_DEBUG "-O0 -g -check all -traceback -fpe0")
endif()

set(CMAKE_Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/mod)

file(GLOB_RECURSE HYDFT_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.f90)

add_library(hydft_lib STATIC ${HYDFT_SOURCES})
target_include_directories(hydft_lib PUBLIC ${FFTW_INCLUDE_DIRS} ${CMAKE_BINARY_DIR}/mod)
target_link_libraries(hydft_lib PUBLIC ${FFTW_LIBRARIES})
if(HYDFT_OPENMP)
  target_link_libraries(hydft_lib PUBLIC OpenMP::OpenMP_Fortran)
endif()

file(GLOB HYDFT_APPS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/app/*.f90)
foreach(app_src ${HYDFT_APPS})
  get_filename_component(app ${app_src} NAME_WE)
  add_executable(${app} ${app_src})
  target_link_libraries(${app} PRIVATE hydft_lib)
endforeach()

enable_testing()
file(GLOB HYDFT_TESTS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/test/test_*.f90)
foreach(t_src ${HYDFT_TESTS})
  get_filename_component(t ${t_src} NAME_WE)
  add_executable(${t} ${t_src})
  target_link_libraries(${t} PRIVATE hydft_lib)
  add_test(NAME ${t} COMMAND ${t} WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
endforeach()
